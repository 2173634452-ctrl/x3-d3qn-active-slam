cmake_minimum_required(VERSION 3.16)
project(x3_rgbd_benchmark)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(cv_bridge REQUIRED)
find_package(OpenCV REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(Pangolin REQUIRED)
set(ORB_ROOT ${CMAKE_CURRENT_SOURCE_DIR}/../third_party/ORB_SLAM3 CACHE PATH "Local ORB-SLAM3 core")
add_executable(x3_rgbd rgbd_node.cpp)
# Match Eigen's SIMD alignment to the ORB-SLAM3 and g2o libraries.
target_compile_options(x3_rgbd PRIVATE -march=native)
target_include_directories(x3_rgbd PRIVATE ${ORB_ROOT} ${ORB_ROOT}/include ${ORB_ROOT}/include/CameraModels ${ORB_ROOT}/Thirdparty/Sophus ${EIGEN3_INCLUDE_DIRS} ${OpenCV_INCLUDE_DIRS} ${Pangolin_INCLUDE_DIRS})
ament_target_dependencies(x3_rgbd rclcpp sensor_msgs nav_msgs std_msgs cv_bridge tf2_ros)
target_link_libraries(x3_rgbd ${ORB_ROOT}/lib/libORB_SLAM3.so ${OpenCV_LIBS} ${Pangolin_LIBRARIES})
